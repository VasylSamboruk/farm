  isOpen: boolean;
  onClose: () => void;

  id: string;
  name: string;
  price: number;
  icon: string;
  profit: number;
  time: string;
  xp: number;
  category: Category;
}

export const ShopModal: React.FC<ShopModalProps> = ({ isOpen, onClose }) => {
  const [activeCategory, setActiveCategory] = useState<Category>('plants');
  const { user } = useAuthStore();

  if (!isOpen) return null;

  const filteredItems = SHOP_ITEMS.filter((item) => item.category === activeCategory);

  const handleBuy = (item: ShopItem) => {
    if ((user?.coins || 0) < item.price) {
      alert('Невистачає монет!');
      return;
    }
    alert(`Ви придбали: ${item.name}!`);
  };

  return (
    <div style={styles.overlay} onClick={onClose}>
      <div style={styles.glassWindow} onClick={(e) => e.stopPropagation()}>
        
        {/* ВЕРХНЯ ШАПКА З ЗАГОЛОВКОМ, РЕСУРСАМИ ТА ЗАКРИТТЯМ */}
        <div style={styles.header}>
          <div style={styles.titleWrapper}>
            <span style={{ fontSize: '22px' }}>🏪</span>
            <h2 style={styles.headerTitle}>МАГАЗИН</h2>
          </div>

          {/* Індикатори Монет та Рубінів */}
          <div style={styles.resourcesContainer}>
            <div style={styles.resourceItem}>
              <span style={styles.resourceIcon}>🪙</span>
              <span style={styles.resourceValue}>{user?.coins || 1000}</span>
            </div>

            <div style={styles.resourceDivider} />

            <div style={styles.resourceItem}>
              <span style={styles.resourceIcon}>💎</span>
              <span style={styles.resourceValue}>25</span>
            </div>
          </div>

          <button style={styles.closeBtn} onClick={onClose}>
            ✕
          </button>
        </div>

        {/* ТАБИ КАТЕГОРІЙ */}
        <div style={styles.tabContainer}>
          <button
            style={activeCategory === 'plants' ? { ...styles.tabBtn, ...styles.activeTabBtn } : styles.tabBtn}
            onClick={() => setActiveCategory('plants')}
          >
            🌱 Рослини
          </button>
          <button
            style={activeCategory === 'trees' ? { ...styles.tabBtn, ...styles.activeTabBtn } : styles.tabBtn}
            onClick={() => setActiveCategory('trees')}
          >
            🌳 Дерева
          </button>
          <button
            style={activeCategory === 'animals' ? { ...styles.tabBtn, ...styles.activeTabBtn } : styles.tabBtn}
            onClick={() => setActiveCategory('animals')}
          >
            🐮 Тварини
          </button>
        </div>

        {/* СІТКА КАРТОК ТОВАРІВ */}
        <div style={styles.gridContainer}>
          {filteredItems.map((item) => (
            <div key={item.id} style={styles.itemCard}>
              
              <div style={styles.cardTitle}>{item.name}</div>

              <div style={styles.iconBox}>
                <span style={styles.cardEmoji}>{item.icon}</span>
              </div>

              <div style={styles.priceBadge}>
                🪙 {item.price}
              </div>

              <div style={styles.statsContainer}>
                <div style={styles.statRow}>
                  <span>Виручка:</span>
                  <span style={{ color: '#81c784', fontWeight: 'bold' }}>+{item.profit}</span>
                </div>
                <div style={styles.statRow}>
                  <span>Готовність:</span>
                  <span>{item.time}</span>
                </div>
                <div style={styles.statRow}>
                  <span>Дає досвід:</span>
                  <span style={{ color: '#b388ff', fontWeight: 'bold' }}>+{item.xp} XP</span>
                </div>
              </div>

              <button style={styles.buyBtn} onClick={() => handleBuy(item)}>
                Купити
              </button>

            </div>
          ))}
        </div>

      </div>
    </div>
  );
};
